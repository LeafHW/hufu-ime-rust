cmake_minimum_required(VERSION 3.21)
project(HuFuSentenceNative LANGUAGES C CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "Use MSVC on Windows (x64 or ARM64); exported C ABI remains GNU-client compatible")
endif()
if(NOT DEFINED HUFU_LLAMA_SOURCE_DIR)
  message(FATAL_ERROR "Run prepare.py to obtain and verify the locked llama.cpp source")
endif()
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_STATIC ON CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
foreach(option COMMON TESTS TOOLS EXAMPLES SERVER APP UI)
  set(LLAMA_BUILD_${option} OFF CACHE BOOL "" FORCE)
endforeach()
add_subdirectory("${HUFU_LLAMA_SOURCE_DIR}" "${CMAKE_BINARY_DIR}/llama.cpp")
add_library(TigerClaw.Sentence.Native SHARED upstream/SentenceQwenNative.cpp upstream/SentenceQwenNative.h)
target_compile_features(TigerClaw.Sentence.Native PRIVATE cxx_std_17)
target_compile_definitions(TigerClaw.Sentence.Native PRIVATE TIGERCLAW_SENTENCE_NATIVE_EXPORTS)
target_compile_options(TigerClaw.Sentence.Native PRIVATE /utf-8 /EHsc)
target_link_libraries(TigerClaw.Sentence.Native PRIVATE llama)
set_target_properties(TigerClaw.Sentence.Native PROPERTIES OUTPUT_NAME "TigerClaw.Sentence.Native")
add_executable(hufu-native-score-probe score_probe.cpp)
target_compile_features(hufu-native-score-probe PRIVATE cxx_std_17)
target_compile_options(hufu-native-score-probe PRIVATE /utf-8 /EHsc)
